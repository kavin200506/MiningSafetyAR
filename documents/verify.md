## 1. Executive Summary & Purpose

The **DGMS Certificate Verification Portal** is a web application designed to allow mine safety inspectors, shift supervisors, and DGMS compliance officers to verify digital safety certificates generated by the **MiningSafetyAR** Unity app.

When a worker completes an AR safety training module and passes the assessment, a cryptographically signed certificate is created and saved to **Firebase Firestore**. Scanning the QR code on a printed or digital certificate opens the web portal:

```text
https://miningsafetyar.web.app/verify?cert=JH-FIRE-922844
```

The webpage fetches the certificate record from Cloud Firestore, validates the signature **in the background** (without displaying technical hashes to the public), and renders a **clean, highly professional, government-standard Digital Safety Certificate**.

## 3. Background Cryptographic Protocol (Hidden from UI)

> [!IMPORTANT]
> Cryptographic HMAC calculations run **entirely in memory in the background**. Internal technical attributes such as raw payload strings, secret keys, and Base64 hashes are **never displayed** on the public user interface.

### Canonical Data Payload Formula
$$\text{RawPayload} = \text{id} \parallel \text{"|"} \parallel \text{workerId} \parallel \text{"|"} \parallel \text{moduleId} \parallel \text{"|"} \parallel \text{score} \parallel \text{"|"} \parallel \text{issuedDate}$$

### Silent Hash Verification (JS)
```javascript
const SECRET_KEY = "DGMS_MINING_SAFETY_SECRET_KEY_2026";

// Silent background verification
function verifyHashSilently(cert) {
    const rawPayload = `${cert.id}|${cert.workerId}|${cert.moduleId}|${cert.score}|${cert.issuedDate}`;
    const computedHash = CryptoJS.HmacSHA256(rawPayload, SECRET_KEY).toString(CryptoJS.enc.Base64);
    return (computedHash === cert.signatureHash);
}
```

        const SECRET_KEY = "DGMS_MINING_SAFETY_SECRET_KEY_2026";
        const FIRESTORE_PROJECT = "minesafetyar";

        window.addEventListener('DOMContentLoaded', () => {
            const params = new URLSearchParams(window.location.search);
            const certId = params.get('cert');
            if (certId) {
                document.getElementById('certInput').value = certId;
                fetchAndVerify(certId);

        function triggerSearch() {
            const val = document.getElementById('certInput').value.trim();
            if (val) fetchAndVerify(val);
        }

        async function fetchAndVerify(certId) {
            showLoading(true);
            hideCert();

            try {
                const cert = await queryFirestore(certId);

                if (!cert) {
                    renderStatus('invalid', '❌ CERTIFICATE NOT FOUND IN REGISTRY');
                    return;
                }

                // Silent Background HMAC Check
                const rawPayload = `${cert.id}|${cert.workerId}|${cert.moduleId}|${cert.score}|${cert.issuedDate}`;
                const computedHash = CryptoJS.HmacSHA256(rawPayload, SECRET_KEY).toString(CryptoJS.enc.Base64);
                const isSignatureValid = (computedHash === cert.signatureHash);

                const isExpired = Date.now() > new Date(cert.expiryDate).getTime();

                // Populate Fields
                document.getElementById('valCertId').innerText = cert.id;
                document.getElementById('valWorkerName').innerText = cert.workerName || "Mining Worker";
                document.getElementById('valWorkerId').innerText = cert.workerId || "W-0000";
                document.getElementById('valModuleTitle').innerText = cert.moduleTitle || cert.moduleId;
                document.getElementById('valScore').innerText = `${cert.score}% (PASSED)`;
                document.getElementById('valIssueDate').innerText = cert.issuedDate;
                document.getElementById('valExpiryDate').innerText = cert.expiryDate;
                document.getElementById('valOrg').innerText = cert.organization || "DGMS Certified Mining Unit";

                // Render Verdict Seal
                if (!isSignatureValid) {
                    renderStatus('invalid', '❌ INVALID CERTIFICATE (SIGNATURE TAMPERED)');
                } else if (isExpired) {
                    renderStatus('expired', '⚠️ CERTIFICATE EXPIRED');
                } else {
                    renderStatus('valid', '✅ OFFICIALLY VERIFIED & VALID');
                }

            } catch (err) {
                console.error("Verification error:", err);
                renderStatus('invalid', '❌ SYSTEM ERROR VERIFYING CERTIFICATE');
            } finally {
                showLoading(false);
            }
        }

        async function queryFirestore(certId) {
            const url = `https://firestore.googleapis.com/v1/projects/${FIRESTORE_PROJECT}/databases/(default)/documents:runQuery`;
            const body = {
                structuredQuery: {
                    from: [{ collectionId: "certificates", allDescendants: true }],
                    where: {
                        fieldFilter: {
                            field: { fieldPath: "id" },
                            op: "EQUAL",
                            value: { stringValue: certId }
                        }
                    }
                }
            };

            const res = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
            if (!res.ok) return null;
            const data = await res.json();
            if (!data || !data[0] || !data[0].document) return null;

            const f = data[0].document.fields;
            return {
                id: f.id?.stringValue || certId,
                workerName: f.workerName?.stringValue || "",
                workerId: f.workerId?.stringValue || "",
                moduleId: f.moduleId?.stringValue || "",
                moduleTitle: f.moduleTitle?.stringValue || "",
                score: f.score?.integerValue || f.score?.stringValue || 0,
                issuedDate: f.issuedDate?.stringValue || "",
                expiryDate: f.expiryDate?.stringValue || "",
                organization: f.organization?.stringValue || "",
                signatureHash: f.signatureHash?.stringValue || ""
            };
        }

        function renderStatus(type, text) {
            const seal = document.getElementById('statusSeal');
            seal.className = `status-seal ${type}`;
            document.getElementById('statusText').innerText = text;
            document.getElementById('certCard').classList.remove('hidden');
            document.getElementById('actionsBar').classList.remove('hidden');
        }

        function showLoading(show) { document.getElementById('loading').classList.toggle('hidden', !show); }
        function hideCert() { document.getElementById('certCard').classList.add('hidden'); document.getElementById('actionsBar').classList.add('hidden'); }
